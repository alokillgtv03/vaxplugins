function withEmbedSid(q, parentHost = '') {
  let s = String(q || '');

  // 1. Nối thêm sid nếu chưa có
  if (!/(?:^|&)sid=/.test(s)) {
    s = `${s}${s.length ? '&' : ''}sid=${encodeURIComponent(EMBED_SID)}`;
  }

  // 2. Nối thêm parent nếu chưa có
  if (!/(?:^|&)parent=/.test(s)) {
    // Ưu tiên truyền tay parentHost, nếu không truyền thì lấy location.hostname
    const host = parentHost || (typeof window !== 'undefined' ? window.location.hostname : '');
    if (host) {
      s = `${s}&parent=${encodeURIComponent(host)}`;
    }
  }

  return s;
}
